def read(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))


def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')


def copy_evidence(folder):
    source = AUTHORITY / EVIDENCE / folder
    target = PRODUCT / EVIDENCE / folder
    for path in source.rglob('*'):
        if not path.is_file():
            continue
        dest = target / path.relative_to(source)
        if dest.exists():
            assert dest.read_bytes() == path.read_bytes(), str(dest)
        else:
            dest.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(path, dest)

ansi = re.compile(r'\x1b\[[0-?]*[ -/]*[@-~]')
diagnosis = []
for target, ordinal, gap in (('darwin-arm64', 1, 7), ('darwin-x64', 1, 59), ('win32-x64', 2, 63)):
    directory = native / 'artifacts' / ('dsh-enhancement-' + target)
    rows = [json.loads(line) for line in (directory / 'terminal-0.jsonl').read_text(encoding='utf-8-sig').splitlines()]
    writes = [(i, e, base64.b64decode(e['data']).decode('utf-8')) for i, e in enumerate(rows)
              if e.get('direction') == 'driver' and e.get('op') == 'write']
    keys = [x for x in writes if x[2] == 'e']
    index, event, _ = keys[-1]
    previous = [x for x in writes if x[0] < index][-1]
    following = next(x for x in writes if x[0] > index)

    def output(first, last):
        return ansi.sub('', ''.join(base64.b64decode(x['data']).decode('utf-8', errors='replace')
                                   for x in rows[first:last] if x.get('direction') == 'host' and x.get('event') == 'output'))

    before = output(0, index)
    after = output(index, following[0])
    last_action_at = before.rfind('<e> 原生改名')
    last_action = before[last_action_at:last_action_at + len('<e> 原生改名 ×')]
    app = [json.loads(line) for line in (directory / 'terminal-0-application.jsonl').read_text(encoding='utf-8-sig').splitlines()]
    previews = [x for x in app if x.get('event') == 'preview' and x.get('action') == 'rename-session']
    mutation_starts = sum(x.get('event') == 'mutation-start' for x in app)
    assert len(keys) == ordinal and previous[2] == 'y' and event['at'] - previous[1]['at'] == gap
    assert last_action == '<e> 原生改名 ×'
    assert '正在核实所选会话的原生操作能力' in after and '原生会话新名称' not in after
    assert mutation_starts == 0 and following[2] == '\x1b'
    assert len(previews) == (2 if target == 'win32-x64' else 0)
    diagnosis.append({
        'target': target, 'failedFormOrdinal': ordinal, 'copyToKeyMs': gap,
        'copyAt': previous[1]['at'], 'keyAt': event['at'], 'cleanupEscapeAt': following[1]['at'],
        'lastRenderedAction': last_action, 'capabilityCheckingNoticeAfterKey': True,
        'freshFormBeforeTimeoutCleanup': False, 'terminalRenamePreviewCount': len(previews),
        'terminalMutationStartCount': mutation_starts,
        'log': 'artifacts/dsh-enhancement-' + target + '/terminal-0.jsonl',
    })
